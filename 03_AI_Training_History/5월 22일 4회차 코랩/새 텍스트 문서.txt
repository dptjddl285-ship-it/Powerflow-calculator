from ultralytics import YOLO

# 1. 's(스몰)' 대신 'm(미디엄)' 모델 투입! (더 깊고 복잡한 패턴을 파악하는 뇌)
model = YOLO('yolo11m.pt')

# 2. 영혼까지 끌어모은 최후의 밸런스 세팅
model.train(
    data=f"{dataset.location}/data.yaml",
    epochs=300,
    patience=100,
    imgsz=1280,
    batch=4,        # [핵심] 뇌가 커졌으므로 메모리 초과(OOM) 방지를 위해 batch를 4로 줄임
    device=0,
    optimizer='auto',
    plots=True,

    # 🎨 사용자님 요청 반영: '섬세한' 훈련 다양성(Augmentation) 추가
    mosaic=0.2,     # 괴물 박스 방지를 위해 1.0은 안 되지만, 0.2(20%)로 약하게 켜서 문맥을 보게 함
    degrees=10.0,   # [요청 반영] 삐뚤어진 사진에 적응하도록 +-10도 회전 허용!
    scale=0.2,      # 크기 변화 20% 허용 (너무 크면 또 괴물 박스 나오니 조심)
    fliplr=0.5,     # 좌우 반전은 기호의 의미를 훼손하지 않으므로 50% 확률로 팍팍 추가!
    hsv_v=0.2       # 조명(밝기) 변화 허용
)
